// ======================= RATE LIMITING =======================
const hits = new Map();
function rateLimit(req, res, next) {
  const ip = req.ip, now = Date.now();
  const rec = hits.get(ip) || { count: 0, reset: now + 60000 };
  if (now > rec.reset) { rec.count = 0; rec.reset = now + 60000; }
  rec.count++; hits.set(ip, rec);
  if (rec.count > 20) return res.status(429).json({ error: 'Too many requests. Wait a minute.' });
  next();
}

// ======================= FORMULA ENGINE ======================
const Calc = {
  depreciationSL({ cost, salvage, life }) {
    if (life <= 0) throw new Error('Useful life must be greater than 0.');
    const annual = (cost - salvage) / life;
    return {
      formula: 'Annual Depreciation = (Cost − Salvage Value) ÷ Useful Life',
      substitution: `= (${fmt(cost)} − ${fmt(salvage)}) ÷ ${life}`,
      steps: [`Cost − Salvage = ${fmt(cost - salvage)}`, `${fmt(cost - salvage)} ÷ ${life} = ${fmt(annual)}`],
      answer: `Annual depreciation = ${fmt(annual)} per year`
    };
  },
  elasticity({ p1, p2, q1, q2 }) {
    const dq = (q2 - q1) / ((q1 + q2) / 2);
    const dp = (p2 - p1) / ((p1 + p2) / 2);
    if (dp === 0) throw new Error('Price did not change — elasticity is undefined.');
    const e = dq / dp;
    const type = Math.abs(e) > 1 ? 'Elastic' : Math.abs(e) < 1 ? 'Inelastic' : 'Unit elastic';
    return {
      formula: 'PED = (ΔQ ÷ avg Q) ÷ (ΔP ÷ avg P)  [midpoint method]',
      substitution: `= ((${q2} − ${q1}) ÷ ${(q1+q2)/2}) ÷ ((${p2} − ${p1}) ÷ ${(p1+p2)/2})`,
      steps: [`ΔQ% = ${(dq*100).toFixed(2)}%`, `ΔP% = ${(dp*100).toFixed(2)}%`, `PED = ${e.toFixed(4)} ≈ ${e.toFixed(2)}`],
      answer: `PED ≈ ${e.toFixed(2)} → Demand is ${type}`
    };
  },
  convert({ value, from, to }) {
    const bases = { binary: 2, octal: 8, decimal: 10, hexadecimal: 16 };
    const fb = bases[from], tb = bases[to];
    const n = parseInt(String(value), fb);
    if (isNaN(n)) throw new Error(`"${value}" is not a valid ${from} number.`);
    const result = n.toString(tb).toUpperCase();
    const steps = [];
    if (from === 'binary') {
      let sum = 0;
      String(value).split('').reverse().forEach((b, i) => { if (b === '1') { sum += 2 ** i; steps.push(`1×2^${i} = ${2**i}`); } });
      steps.push(`Sum = ${n}`);
    }
    if (from === 'decimal' && to === 'binary') {
      let x = n; const bits = [];
      while (x > 0) { bits.unshift(x % 2); steps.push(`${x} ÷ 2 = ${Math.floor(x/2)} remainder ${x%2}`); x = Math.floor(x / 2); }
      steps.push(`Read remainders bottom-up → ${bits.join('')}`);
    }
    return { formula: `${from} → ${to} (base ${fb} → base ${tb})`, substitution: `${value} base ${fb}`,
      steps: steps.length ? steps : ['Direct conversion'], answer: `= ${result}` };
  },
  truthTable({ gate }) {
    const g = gate.toUpperCase();
    const ops = { AND:(a,b)=>a&b, OR:(a,b)=>a|b, XOR:(a,b)=>a^b, NAND:(a,b)=>~(a&b)&1, NOR:(a,b)=>~(a|b)&1, XNOR:(a,b)=>~(a^b)&1 };
    if (g === 'NOT') return { formula:'NOT gate (inverter)', substitution:'Y = NOT A', steps:[],
      answer:'A | Y\n0 | 1\n1 | 0' };
    if (!ops[g]) throw new Error('Supported gates: AND OR NOT NAND NOR XOR XNOR');
    const rows = [[0,0],[0,1],[1,0],[1,1]].map(([a,b]) => `${a} ${b} | ${ops[g](a,b)}`);
    return { formula:`${g} gate truth table`, substitution:`Boolean: Y = A ${g==='AND'?'·':g==='OR'?'+':g} B`,
      steps:['Evaluate the gate for all 4 input combinations of A and B'],
      answer:`A B | Y\n${rows.join('\n')}` };
  }
};

function fmt(n) {
  return Number.isInteger(n) ? n.toLocaleString('en-US') : n.toLocaleString('en-US', {maximumFractionDigits:2});
}

function tryCalculation(message) {
  const m = message.toLowerCase();

  let depMatch = m.match(/(?:cost|price)?\s*(?:rs\.?|lkr)?\s*([\d,]+).*?(?:salvage|residual|scrap)[^\d]*([\d,]+).*?(?:life|years)[^\d]*(\d+)/s)
    || (m.includes('depreciation') && m.match(/([\d,]+)[^\d]+(?:salvage|residual)[^\d]*([\d,]+)[^\d]*(?:life|years)[^\d]*(\d+)/s));
  if (m.includes('depreciation') && depMatch) {
    const [, c, s, l] = depMatch;
    return Calc.depreciationSL({ cost:+c.replace(/,/g,''), salvage:+s.replace(/,/g,''), life:+l });
  }

  if (m.includes('elastic')) {
    const nums = [...m.matchAll(/[\d.]+/g)].map(x => parseFloat(x[0]));
    if (nums.length >= 4) return Calc.elasticity({ p1:nums[0], p2:nums[1], q1:nums[2], q2:nums[3] });
  }

  const bases = ['binary','octal','decimal','hexadecimal','hex'];
  const conv = m.match(/([\w.]+)\s*(binary|octal|decimal|hexadecimal|hex)\s*(?:to|into)\s*(binary|octal|decimal|hexadecimal|hex)/)
        || m.match(/(binary|octal|decimal|hexadecimal|hex)\s*(?:to|into)\s*(binary|octal|decimal|hexadecimal|hex)\s*(?:of\s*)?([\w.]+)/);
  if (conv) {
    let value, from, to;
    if (!bases.includes(conv[1])) { value = conv[1]; from = conv[2]; to = conv[3]; }
    else { from = conv[1]; to = conv[2]; value = conv[3]; }
    from = from === 'hex' ? 'hexadecimal' : from;
    to = to === 'hex' ? 'hexadecimal' : to;
    if (from !== to && value) return Calc.convert({ value, from, to });
  }

  const tt = m.match(/truth table(?:\s*(?:for|of)\s*|\s+)?(and|or|not|nand|nor|xor|xnor)\b/) || m.match(/\b(and|or|not|nand|nor|xor|xnor)\s+gate\b.*truth/);
  if (tt) return Calc.truthTable({ gate: tt[1] });
  if (m.includes('truth table')) {
    const g = (m.match(/(and|or|not|nand|nor|xor|xnor)/) || [])[1];
    if (g) return Calc.truthTable({ gate: g });
  }
  return null;
}

// ======================= SUBJECT ROUTING =======================
const SUBJECT_HINTS = {
  ICT: ["logic gate","binary","two's complement","database","normalization","sql","python","algorithm","flowchart","hexadecimal","boolean","truth table"],
  Accounting: ["journal","ledger","trial balance","depreciation","inventory","bank reconciliation","financial statement","double entry","capital"],
  Economics: ["demand","supply","elasticity","inflation","national income","opportunity cost","market equilibrium","monetary policy","fiscal"],
  "Business Studies": ["entrepreneur","management","marketing","human resource","business plan","leadership","organizing","business environment"]
};
function detectSubject(text) {
  const t = text.toLowerCase();
  let best = null, bestScore = 0;
  for (const [s, words] of Object.entries(SUBJECT_HINTS)) {
    const score = words.filter(w => t.includes(w)).length;
    if (score > bestScore) { bestScore = score; best = s; }
  }
  return best || 'General';
}

  Accounting: "You are an expert Sri Lankan G.C.E. A/L Accounting tutor. Always show debit/credit reasoning, the accounting equation check, formulas, substitutions, and step-by-step workings. Never invent LKAS/SLFRS rules.",
  Economics: "You are an expert Sri Lankan G.C.E. A/L Economics tutor. Give definitions, equations, worked numerical examples, graph interpretations, and economic intuition.",
  "Business Studies": "You are an expert Sri Lankan G.C.E. A/L Business Studies tutor. Give definitions, structured explanations, advantages/disadvantages, and practical examples.",

};

// ======================= API =======================
app.post('/api/chat', rateLimit, async (req, res) => {
  const { message, subject, grade, language, image } = req.body || {};
  if (!message || typeof message !== 'string' || message.length > 4000)
    return res.status(400).json({ error: 'Invalid message.' });

  const subj = (subject && subject !== 'auto') ? subject : detectSubject(message);

  // 1) Deterministic calculation engine — exact answers, no AI guessing
  if (!image) {
    try {
      const calc = tryCalculation(message);
      if (calc) {
        const out = `⚙️ Solved with the Formula Engine (exact calculation)\n\nFormula: ${calc.formula}\nSubstitution: ${calc.substitution}\n\nSteps:\n${calc.steps.map((s,i)=>`${i+1}. ${s}`).join('\n')}\n\n✅ Final answer: ${calc.answer}`;
        return res.json({ answer: out, subject: subj, engine: 'formula' });
      }
    } catch (e) {
      return res.json({ answer: `⚠️ Formula Engine could not compute this: ${e.message}\nPlease check the numbers in your question and try again.`, subject: subj, engine: 'formula' });
    }
  }

  // 2) Otherwise → Gemini
  if (!GEMINI_KEY)
    return res.status(500).json({ error: 'Server misconfiguration: GEMINI_API_KEY is not set.' });

  const langNote = {
    'Sinhala': 'Answer in clear Sinhala Unicode. Keep standard technical terms in English where natural.',
    'Singlish': 'Answer in Singlish (romanized Sinhala).',
    'English': 'Answer in clear English.'
  }[language] || "Match the language of the student's question.";

  const system = SYSTEM_BY_SUBJECT[subj] + ' ' + langNote +
    (grade ? ` The student is in ${grade}.` : '') +
    ' Structure: direct answer, definition, explanation, step-by-step working where needed, an example, and exam tips when useful.';

  const parts = [{ text: message }];
  if (image) parts.push({ inline_data: { mime_type: 'image/jpeg', data: image } });

  const body = {
    system_instruction: { parts: [{ text: system }] },
    contents: [{ parts }],
    generationConfig: { temperature: 0.4, maxOutputTokens: 2048 }
  };

  async function callOnce() {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), 30000);
    try {
      return await fetch(`${API_URL}?key=${GEMINI_KEY}`, {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body), signal: ctrl.signal
      });
    } finally { clearTimeout(timer); }
  }

  try {
    let r = await callOnce();
    if (r.status >= 500) r = await callOnce();
    if (r.status === 429) return res.status(429).json({ error: 'AI quota exceeded. Try again shortly.' });
    if (r.status === 400 || r.status === 403) return res.status(502).json({ error: 'AI credentials invalid. Check server GEMINI_API_KEY.' });
    if (!r.ok) return res.status(502).json({ error: `AI provider error (${r.status}).` });
    const data = await r.json();
    const text = data?.candidates?.[0]?.content?.parts?.map(p => p.text).join('');
    if (!text) return res.status(502).json({ error: 'AI returned an empty/malformed response.' });
    res.json({ answer: text, subject: subj, engine: 'gemini' });
  } catch (e) {
    res.status(504).json({ error: e.name === 'AbortError' ? 'Request timed out.' : 'Network error contacting AI provider.' });
  }
});
